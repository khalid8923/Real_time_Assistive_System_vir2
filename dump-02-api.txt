export const runtime = "nodejs";

interface LoginBody {

const attempts = new Map<string, { count: number; lockedUntil: number }>();
const MAX_ATTEMPTS = 5;
const LOCK_MS = 5 * 60 * 1000;

function getClientKey(request: NextRequest): string {
  const ip =
    request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
    request.headers.get("x-real-ip") ||
    "unknown";
  return ip;
}

export async function POST(request: NextRequest) {
  const adminPassword = process.env.ADMIN_PASSWORD?.trim();
  if (!adminPassword) {
    console.error("[admin/login] ADMIN_PASSWORD is not set");
    return NextResponse.json(
      { error: "Server misconfiguration." },
      { status: 500 }
    );
  }

  const clientKey = getClientKey(request);
  const record = attempts.get(clientKey);

  if (record && record.lockedUntil > Date.now()) {
    const secondsLeft = Math.ceil((record.lockedUntil - Date.now()) / 1000);
    return NextResponse.json(
      { error: `محاولات كثيرة. حاول بعد ${secondsLeft} ثانية.` },
      { status: 429 }
    );
  }

  let body: LoginBody;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ error: "طلب غير صالح." }, { status: 400 });
  }

  const { password } = body;
  if (typeof password !== "string" || password.trim().length === 0) {
    return NextResponse.json({ error: "الباسورد مطلوب." }, { status: 400 });
  }

  const isValid = await verifyAdminPassword(password.trim());

  if (!isValid) {
    const current = attempts.get(clientKey) ?? { count: 0, lockedUntil: 0 };
    current.count += 1;

    if (current.count >= MAX_ATTEMPTS) {
      current.lockedUntil = Date.now() + LOCK_MS;
      current.count = 0;
      attempts.set(clientKey, current);
      console.warn(`[admin/login] ✗ ${clientKey} locked`);

      return NextResponse.json(
        { error: "محاولات كثيرة. تم القفل مؤقتاً لـ 5 دقايق." },
        { status: 429 }
      );
    }

    attempts.set(clientKey, current);
    console.warn(
      `[admin/login] ✗ ${clientKey} wrong password (${current.count}/${MAX_ATTEMPTS})`
    );

    return NextResponse.json({ error: "باسورد خاطئ." }, { status: 401 });
  }

  attempts.delete(clientKey);

  const response = NextResponse.json({ ok: true });
  response.cookies.set("admin_mode", "true", {
    httpOnly: true,
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
    path: "/",
    maxAge: 60 * 60 * 12,
  });

  console.log(`[admin/login] ✓ ${clientKey} logged in`);
  return response;
}

function getHakimKeys(): string[] {
  return [

    process.env.HAKIM_API_KEY_2?.trim(),
    process.env.HAKIM_API_KEY_3?.trim(),

// Simple round-robin counter — يتصفّر كل restart
let hakimRoundRobin = 0;

async function transcribeWithHakim(audio: Blob): Promise<string> {
  const keys = getHakimKeys();
  if (keys.length === 0) return "";

  // ابدأ من المفتاح التالي في الـ round-robin
  const startIdx = hakimRoundRobin % keys.length;
  hakimRoundRobin = (hakimRoundRobin + 1) % keys.length;

  // جرّب كل المفاتيح بالترتيب (يبدأ من الـ round-robin)
  for (let i = 0; i < keys.length; i++) {
    const idx = (startIdx + i) % keys.length;
    const apiKey = keys[idx];

    const formData = new FormData();
    formData.append("file", audio, "chunk.webm");
    formData.append("model", "hakim-arab-v2");
    formData.append("language", "ar");
    formData.append("response_format", "json");

    try {
      const res = await fetch(
        "https://api.tryhakim.ai/v1/audio/transcriptions",
        {
          method: "POST",
          headers: { Authorization: `Bearer ${apiKey}` },
          body: formData,
        }
      );

      if (!res.ok) {
        const errText = await res.text();
        console.error(`[hakim #${idx + 1}] ${res.status}:`, errText.slice(0, 150));

        // 429 أو 401 → جرّب المفتاح التالي
        if (res.status === 429 || res.status === 401 || res.status === 403) {
          continue;
        }
        // خطأ تاني → اطلع
        return "";
      }

      const data = (await res.json()) as {
        text?: string;
        data?: { text?: string };
      };

      const text = (data.text ?? data.data?.text ?? "").trim();
      if (text.length > 0) {
        console.log(`[smart] Hakim key #${idx + 1} returned ${text.length}c`);
        return text;
      }

      // نص فاضي → جرّب المفتاح التالي
      console.log(`[smart] Hakim key #${idx + 1} returned empty`);
    } catch (err) {
      console.warn(`[hakim #${idx + 1}] threw:`, err);
      continue;
    }
  }

  return "";
}

async function transcribeWithGroq(audio: Blob): Promise<string> {
  const apiKey = process.env.GROQ_API_KEY?.trim();
  if (!apiKey) return "";

  const formData = new FormData();
  formData.append("file", audio, "chunk.webm");
  formData.append("model", "whisper-large-v3-turbo");
  formData.append("response_format", "json");
  formData.append("temperature", "0");
  formData.append("language", "ar");
  formData.append(
    "prompt",
    "دكتور جامعي يشرح بالعربي المصري. احتفظ بالمصطلحات الإنجليزية."
  );

  const res = await fetch(
    "https://api.groq.com/openai/v1/audio/transcriptions",
    {
      method: "POST",
      headers: { Authorization: `Bearer ${apiKey}` },
      body: formData,
    }
  );

  if (!res.ok) {
    const errText = await res.text();
    console.error(`[groq] ${res.status}:`, errText.slice(0, 200));
    return "";
  }

  const data = (await res.json()) as { text?: string };
  return (data.text ?? "").trim();
}

      sql: `SELECT createdAt, wordCount
            FROM saved_lectures
            WHERE userId = ? AND createdAt >= ?
            ORDER BY createdAt ASC`,
      args: [userId, thirtyDaysAgo],
    });

    for (const row of lecturesResult.rows) {
      const d = new Date(Number(row.createdAt));

      cur.words += Number(row.wordCount || 0);

      date: string;
      label: string;
      lectures: number;
      words: number;

interface CausalPayload {
  loops?: Array<{
    title?: string;
    subtitle?: string;
    nodes?: Array<{
      id?: string;
      label?: string;
      detail?: string;
      category?: string;
    }>;
    edges?: Array<{
      from?: string;
      to?: string;
      sign?: string;
      evidence?: string;
      label?: string;
      quote?: string;
    }>;
    sourceQuote?: string;
  }>;
}

function isValid(d: unknown): d is CausalPayload {
  if (typeof d !== "object" || d === null) return false;
  return Array.isArray((d as Record<string, unknown>).loops);
}

function detectLoops(
  nodeIds: string[],
  edges: Array<{ from: string; to: string; sign: "positive" | "negative" }>
): Array<{
  id: string;
  type: "reinforcing" | "balancing";
  nodeIds: string[];
}> {
  const detected: Array<{
    id: string;
    type: "reinforcing" | "balancing";
    nodeIds: string[];
  }> = [];

  const adj = new Map<string, Array<{ to: string; sign: number }>>();
  for (const e of edges) {
    if (!adj.has(e.from)) adj.set(e.from, []);
    adj.get(e.from)!.push({ to: e.to, sign: e.sign === "positive" ? 1 : -1 });
  }

  const seen = new Set<string>();

  // Simple cycle detection via DFS
  function dfs(
    start: string,
    current: string,
    path: string[],
    signProduct: number,
    visitedInPath: Set<string>
  ) {
    if (path.length > 6) return;
    const neighbors = adj.get(current) ?? [];
    for (const { to, sign } of neighbors) {
      if (to === start && path.length >= 2) {
        // Found a cycle
        const key = [...path].sort().join("|");
        if (!seen.has(key)) {
          seen.add(key);
          detected.push({
            id: `loop-${detected.length + 1}`,
            type: signProduct * sign > 0 ? "reinforcing" : "balancing",
            nodeIds: [...path],
          });
        }
      } else if (!visitedInPath.has(to)) {
        visitedInPath.add(to);
        dfs(start, to, [...path, to], signProduct * sign, visitedInPath);
        visitedInPath.delete(to);
      }
    }
  }

  for (const start of nodeIds) {
    dfs(start, start, [start], 1, new Set([start]));
  }

  return detected.slice(0, 4);
}

interface DiagramPayload {
  shape?: string;
  title?: string;
  subtitle?: string;
  nodes?: Array<{
    id?: string;
    label?: string;
    detail?: string;
    icon?: string;
  }>;
  relations?: Array<{
    from?: string;
    to?: string;
    type?: string;
    label?: string;
  }>;
  emphasis?: string[];
  sourceQuote?: string;
}

function isValid(d: unknown): d is DiagramPayload {
  if (typeof d !== "object" || d === null) return false;
  const o = d as Record<string, unknown>;
  return typeof o.shape === "string" && Array.isArray(o.nodes);
}

export async function POST(request: NextRequest) {
  const ip = getClientIp(request);
  if (!rateLimit(ip, 10, 60_000).ok) {
    return NextResponse.json({ error: "Too many requests." }, { status: 429 });
  }

  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ error: "Invalid JSON." }, { status: 400 });
  }

  const rawText = (body as Record<string, unknown>)?.text;
  if (typeof rawText !== "string" || rawText.trim().length < 30) {
    return NextResponse.json({ error: "Text too short." }, { status: 400 });
  }

  const text = rawText.trim().slice(0, 4000);

  const result = await generateText({
    systemPrompt: SYSTEM_PROMPT,
    messages: [{ role: "user", content: `Text:\n\n${text}` }],
    temperature: 0.2,
    maxTokens: 2000,
    jsonMode: true,
    prefer: "nvidia",
  });

  if (!result.ok || !result.text) {
    return NextResponse.json(
      { error: result.error || "Generation failed." },
      { status: 502 }
    );
  }

  const parsed = parseJsonResponse(result.text);
  if (!parsed || !isValid(parsed)) {
    return NextResponse.json({ error: "Invalid schema." }, { status: 502 });
  }

  // Validate shape
  const shape: DiagramShape = ALLOWED_SHAPES.includes(parsed.shape ?? "")
    ? (parsed.shape as DiagramShape)
    : "sequence";

  const now = Date.now();

  // Sanitize nodes
  const nodes = (parsed.nodes ?? [])
    .slice(0, 7)
    .map((n, i) => {
      const icon = n.icon && ALLOWED_ICONS.includes(n.icon) ? n.icon : "circle";
      return {
        id: String(n.id ?? `n${i + 1}`),
        label: String(n.label ?? "").slice(0, 60),
        detail: n.detail ? String(n.detail).slice(0, 120) : undefined,
        icon,
      };
    })
    .filter((n) => n.label.length > 0);

  if (nodes.length < 3) {
    return NextResponse.json(
      { error: "Not enough nodes to render a diagram." },
      { status: 200, headers: { "X-Empty": "1" } }
    );
  }

  const nodeIds = new Set(nodes.map((n) => n.id));

  const relations = (parsed.relations ?? [])
    .filter(
      (r) =>
        typeof r.from === "string" &&
        typeof r.to === "string" &&
        nodeIds.has(r.from) &&
        nodeIds.has(r.to)
    )
    .slice(0, 20)
    .map((r) => ({
      from: String(r.from),
      to: String(r.to),
      type: ["next", "part_of", "opposite", "contains"].includes(r.type ?? "")
        ? (r.type as "next" | "part_of" | "opposite" | "contains")
        : "next",
      label: r.label ? String(r.label).slice(0, 40) : undefined,
    }));

  const emphasis = Array.isArray(parsed.emphasis)
    ? parsed.emphasis.filter((id) => typeof id === "string" && nodeIds.has(id)).slice(0, 3)
    : [];

  const diagram: GeneratedDiagram = {
    id: `diag-${now}`,
    shape,
    title: String(parsed.title ?? "رسم توضيحي"),
    subtitle: parsed.subtitle ? String(parsed.subtitle) : undefined,
    nodes,
    relations,
    emphasis,
    sourceQuote: parsed.sourceQuote ? String(parsed.sourceQuote) : undefined,
    detectedAt: now,
  };

  console.log(
    `[visualize-this] OK via "${result.provider}" | shape=${shape} | nodes=${nodes.length}`
  );

  return NextResponse.json({ diagram }, { status: 200 });
}
