import {
  normalizeError,
  createError,
  categoryFromStatus,
  CATEGORY_META,
  type AppError,
} from "@/lib/errors";

export interface RequestOptions extends RequestInit {
  timeoutMs?: number;
  retries?: number;
  showToast?: boolean;
  onError?: (err: AppError) => void;
}

const DEFAULT_TIMEOUT = 45_000;
const DEFAULT_RETRIES = 2;
const RETRY_BASE_DELAY = 600;

/* ------------------------------------------------------------------ */

function sleep(ms: number): Promise<void> {
  return new Promise((r) => setTimeout(r, ms));
}

function generateRequestId(): string {
  return `req-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
}

function isRetryableStatus(status: number): boolean {
  return (
    status === 408 ||
    status === 429 ||
    status === 502 ||
    status === 503 ||
    status === 504
  );
}

/* ------------------------------------------------------------------ */

export async function apiRequest<T = unknown>(
  input: RequestInfo | URL,
  options: RequestOptions = {}
): Promise<{ ok: true; data: T } | { ok: false; error: AppError }> {
  const {
    timeoutMs = DEFAULT_TIMEOUT,
    retries = DEFAULT_RETRIES,
    showToast = false,
    onError,
    ...fetchOptions
  } = options;

  const requestId = generateRequestId();
  let lastError: AppError | null = null;

  // Offline check
  if (typeof navigator !== "undefined" && navigator.onLine === false) {
    const err = createError("offline", { requestId });
    if (showToast) await showErrorToast(err);
    onError?.(err);
    return { ok: false, error: err };
  }

  for (let attempt = 0; attempt <= retries; attempt++) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeoutMs);

    try {
      const response = await fetch(input, {
        ...fetchOptions,
        signal: controller.signal,
        headers: {
          "X-Request-Id": requestId,
          ...(fetchOptions.headers ?? {}),
        },
      });

      clearTimeout(timer);

      if (response.ok) {
        const contentType = response.headers.get("content-type") ?? "";
        const data = contentType.includes("application/json")
          ? ((await response.json()) as T)
          : ((await response.text()) as unknown as T);
        return { ok: true, data };
      }

      // فشل HTTP
      let errBody: { error?: string } = {};
      try {
        errBody = await response.json();
      } catch {
        /* ignore */
      }

      const category = categoryFromStatus(response.status);
      const err = createError(category, {
        code: `HTTP_${response.status}`,
        httpStatus: response.status,
        technicalMessage: errBody.error ?? response.statusText,
        requestId,
        userMessage: errBody.error || CATEGORY_META[category].description,
      });

      lastError = err;

      if (attempt < retries && isRetryableStatus(response.status)) {
        const delay = RETRY_BASE_DELAY * Math.pow(2, attempt);
        console.warn(
          `[api] retry ${attempt + 1}/${retries} after ${delay}ms (status ${response.status})`
        );
        await sleep(delay);
        continue;
      }

      if (showToast) await showErrorToast(err);
      onError?.(err);
      return { ok: false, error: err };
    } catch (thrown) {
      clearTimeout(timer);
      const err = normalizeError(thrown, requestId);
      lastError = err;

      if (attempt < retries && err.retryable) {
        const delay = RETRY_BASE_DELAY * Math.pow(2, attempt);
        console.warn(
          `[api] retry ${attempt + 1}/${retries} after ${delay}ms (${err.code})`
        );
        await sleep(delay);
        continue;
      }

      if (showToast) await showErrorToast(err);
      onError?.(err);
      return { ok: false, error: err };
    }
  }

  const final = lastError ?? createError("unknown", { requestId });
  if (showToast) await showErrorToast(final);
  onError?.(final);
  return { ok: false, error: final };
}

/* ------------------------------------------------------------------ */

async function showErrorToast(err: AppError): Promise<void> {
  try {
    const { toastError } = await import("@/lib/error-toast");
    toastError(err);
  } catch {
    /* ignore */
  }
}

/* ------------------------------------------------------------------ */

export async function apiGet<T = unknown>(url: string, opts?: RequestOptions) {
  return apiRequest<T>(url, { ...opts, method: "GET" });
}

export async function apiPost<T = unknown>(
  url: string,
  body: unknown,
  opts?: RequestOptions
) {
  return apiRequest<T>(url, {
    ...opts,
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      ...(opts?.headers ?? {}),
    },
    body: JSON.stringify(body),
  });
}

export async function apiPostForm<T = unknown>(
  url: string,
  formData: FormData,
  opts?: RequestOptions
) {
  return apiRequest<T>(url, {
    ...opts,
    method: "POST",
    body: formData,
  });
}

import { toast } from "sonner";

function getIcon(category: ErrorCategory): string {
  const icons: Record<ErrorCategory, string> = {
    network: "📡",
    timeout: "⏱️",
    offline: "🔌",
    auth: "🔐",
    permission: "🚫",
    notFound: "🔍",
    validation: "⚠️",
    rateLimit: "⏳",
    server: "🔧",
    provider: "🤖",
    storage: "💾",
    unknown: "❗",
  };
  return icons[category] ?? "❗";
}

/* ------------------------------------------------------------------ */
/* Error toast                                                        */
/* ------------------------------------------------------------------ */

export function toastError(err: AppError): void {
  const icon = getIcon(err.category);
  toast.error(`${icon} ${err.title}`, {
    description: err.hint
      ? `${err.userMessage}\n${err.hint}`
      : err.userMessage,
    duration: err.retryable ? 7000 : 5500,
  });
}

export function toastSuccess(message: string, description?: string): void {
  toast.success(`✅ ${message}`, { description, duration: 3500 });
}

export function toastInfo(message: string, description?: string): void {
  toast.info(`ℹ️ ${message}`, { description, duration: 4000 });
}

export function toastWarning(message: string, description?: string): void {
  toast.warning(`⚠️ ${message}`, { description, duration: 5500 });
}

/* ------------------------------------------------------------------ */
/* HTTP status → category                                             */
/* ------------------------------------------------------------------ */

export function categoryFromStatus(status: number): ErrorCategory {
  if (status === 401) return "auth";
  if (status === 403) return "permission";
  if (status === 404) return "notFound";
  if (status === 408 || status === 504) return "timeout";
  if (status === 413 || status === 422) return "validation";
  if (status === 429) return "rateLimit";
  if (status === 500 || status === 502 || status === 503) return "server";
  if (status >= 400 && status < 500) return "validation";
  if (status >= 500) return "server";
  return "unknown";
}

/* ------------------------------------------------------------------ */
/* Create error                                                       */
/* ------------------------------------------------------------------ */

export function createError(
  category: ErrorCategory,
  options?: {
    code?: string;
    technicalMessage?: string;
    httpStatus?: number;
    requestId?: string;
    userMessage?: string;
  }
): AppError {
  const meta = CATEGORY_META[category];
  return {
    category,
    code: options?.code ?? category.toUpperCase(),
    title: meta.title,
    userMessage: options?.userMessage ?? meta.description,
    hint: meta.hint,
    technicalMessage: options?.technicalMessage ?? "",
    httpStatus: options?.httpStatus,
    requestId: options?.requestId,
    retryable: ["network", "timeout", "server", "provider", "rateLimit"].includes(
      category
    ),
    timestamp: Date.now(),
  };
}

/* ------------------------------------------------------------------ */
/* Normalize any thrown error                                         */
/* ------------------------------------------------------------------ */

export function normalizeError(err: unknown, requestId?: string): AppError {
  // Already AppError
  if (
    err &&
    typeof err === "object" &&
    "category" in err &&
    "title" in err &&
    "userMessage" in err
  ) {
    return err as AppError;
  }

  // DOMException (abort)
  if (err instanceof DOMException) {
    if (err.name === "AbortError") {
      return createError("timeout", {
        code: "REQUEST_ABORTED",
        technicalMessage: err.message,
        requestId,
      });
    }
    return createError("network", {
      code: "DOM_EXCEPTION",
      technicalMessage: `${err.name}: ${err.message}`,
      requestId,
    });
  }

  // TypeError — network error
  if (err instanceof TypeError) {
    const offline =
      typeof navigator !== "undefined" && navigator.onLine === false;
    return createError(offline ? "offline" : "network", {
      code: "NETWORK_ERROR",
      technicalMessage: err.message,
      requestId,
    });
  }

  // Error
  if (err instanceof Error) {
    return createError("unknown", {
      code: "GENERIC_ERROR",
      technicalMessage: err.message,
      requestId,
    });
  }

  // غير معروف
  return createError("unknown", {
    code: "UNKNOWN_THROWN",
    technicalMessage: String(err),
    requestId,
  });
}

const cache = new Map<string, { count: number; resetAt: number }>();

// ✅ Cleanup كل 5 دقايق (unref عشان مش يمنع الخروج)
if (typeof setInterval !== "undefined") {
  const timer = setInterval(() => {
    const now = Date.now();
    for (const [key, record] of cache.entries()) {
      if (record.resetAt < now) cache.delete(key);
    }
  }, 5 * 60 * 1000);
  if (typeof timer === "object" && "unref" in timer) {
    (timer as { unref: () => void }).unref();
  }
}

export function rateLimit(
  key: string,
  max: number = 30,
  windowMs: number = 60_000
): { ok: boolean; remaining: number; resetIn: number } {
  const now = Date.now();
  const record = cache.get(key);

  if (!record || record.resetAt < now) {
    cache.set(key, { count: 1, resetAt: now + windowMs });
    return { ok: true, remaining: max - 1, resetIn: windowMs };
  }

  if (record.count >= max) {
    return { ok: false, remaining: 0, resetIn: record.resetAt - now };
  }

  record.count++;
  return {
    ok: true,
    remaining: max - record.count,
    resetIn: record.resetAt - now,
  };
}

export function getClientIp(request: Request): string {
  const forwarded = request.headers.get("x-forwarded-for");
  if (forwarded) return forwarded.split(",")[0].trim();
  return request.headers.get("x-real-ip") || "unknown";
}

export const safeStorage = {
  get(key: string): string | null {
    if (typeof window === "undefined") return null;
    try {
      return window.localStorage.getItem(key);
    } catch {
      return null;
    }
  },

  set(key: string, value: string): boolean {
    if (typeof window === "undefined") return false;
    try {
      window.localStorage.setItem(key, value);
      return true;
    } catch (err) {
      if (err instanceof DOMException && err.name === "QuotaExceededError") {
        console.warn("[safe-storage] quota exceeded");
      }
      return false;
    }
  },

  remove(key: string): void {
    if (typeof window === "undefined") return;
    try {
      window.localStorage.removeItem(key);
    } catch {
      /* ignore */
    }
  },

  getJSON<T>(key: string, fallback: T): T {
    const raw = this.get(key);
    if (!raw) return fallback;
    try {
      return JSON.parse(raw) as T;
    } catch {
      return fallback;
    }
  },

  setJSON(key: string, value: unknown): boolean {
    try {
      return this.set(key, JSON.stringify(value));
    } catch {
      return false;
    }
  },
};

async function tryNvidia(options: GenerateOptions): Promise<GenerateResult> {
  const apiKeys = getNvidiaKeys();
  if (apiKeys.length === 0) {
    return { ok: false, error: FRIENDLY_ERRORS.nvidiaMissing };
  }

  const messages = [
    { role: "system", content: options.systemPrompt },
    ...options.messages.map((m) => ({ role: m.role, content: m.content })),
  ];

  let lastErr = "";

  for (let keyIdx = 0; keyIdx < apiKeys.length; keyIdx++) {
    for (const model of NVIDIA_MODELS) {
      try {
        const body: Record<string, unknown> = {
          model,
          messages,
          temperature: options.temperature ?? 0.5,
          max_tokens: options.maxTokens ?? 2048,
          stream: false,
        };
        if (options.jsonMode) {
          body.response_format = { type: "json_object" };
        }

        const res = await fetch(NVIDIA_ENDPOINT, {
          method: "POST",
          headers: {
            Authorization: `Bearer ${apiKeys[keyIdx]}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify(body),
        });

        if (!res.ok) {
          lastErr = `NVIDIA ${res.status}`;
          if (res.status === 429) break; // جرّب المفتاح التالي
          if (res.status === 404 || res.status === 400) continue;
          continue;
        }

        const data = (await res.json()) as {
          choices?: Array<{ message?: { content?: string } }>;
        };
        const text = data.choices?.[0]?.message?.content ?? "";
        if (text.trim()) {
          console.log(`[ai] NVIDIA OK | key #${keyIdx + 1} | model "${model}"`);
          return {
            ok: true,
            text,
            provider: "nvidia",
            model: `${model} (key #${keyIdx + 1})`,
          };
        }
      } catch (err) {
        lastErr = err instanceof Error ? err.message : "unknown";
        continue;
      }
    }
  }

  return { ok: false, error: friendlyError(lastErr) };
}

async function tryGemini(options: GenerateOptions): Promise<GenerateResult> {
  const apiKeys = getGeminiKeys();

      try {
        const response = await ai.models.generateContent({
          model,
          contents,
          config: {
            systemInstruction: options.systemPrompt,
            temperature: options.temperature ?? 0.5,
            maxOutputTokens: options.maxTokens ?? 2048,
          },
        });

        const text = response.text ?? "";
        if (text.trim()) {
          console.log(`[ai] Gemini OK | key #${keyIdx + 1} | model "${model}"`);
          return {
            ok: true,
            text,
            provider: "gemini",
            model: `${model} (key #${keyIdx + 1})`,
          };
        }
      } catch (err) {
        lastErr = err instanceof Error ? err.message : "unknown";
        continue;
      }
    }
  }

  return { ok: false, error: friendlyError(lastErr) };
}

export async function generateText(
  options: GenerateOptions
): Promise<GenerateResult> {
  const order: AIProvider[] =
    options.prefer === "gemini"
      ? ["gemini", "nvidia", "groq"]
      : options.prefer === "groq"
      ? ["groq", "nvidia", "gemini"]
      : ["nvidia", "groq", "gemini"];

  for (const provider of order) {
    const result =
      provider === "nvidia"
        ? await tryNvidia(options)
        : provider === "groq"
        ? await tryGroq(options)
        : await tryGemini(options);

    if (result.ok) return result;
    console.warn(`[ai] ${provider} failed`);
  }

  return { ok: false, error: FRIENDLY_ERRORS.allFailed };
}

export function parseJsonResponse<T = unknown>(raw: string): T | null {
  try {
    const cleaned = raw
      .trim()
      .replace(/^```(?:json)?\s*/i, "")
      .replace(/\s*```$/i, "")
      .trim();
    return JSON.parse(cleaned) as T;
  } catch {
    return null;
  }
}

function withTimeout<T>(promise: Promise<T>, ms = QUERY_TIMEOUT_MS): Promise<T> {
  return Promise.race([
    promise,
    new Promise<T>((_, reject) =>
      setTimeout(
        () => reject(new Error(`DB query timed out after ${ms}ms`)),
        ms
      )
    ),
  ]);
}

export async function safeExecute(
  db: Client,
  query: { sql: string; args?: unknown[] }
) {
  return withTimeout(
    db.execute({ sql: query.sql, args: (query.args ?? []) as never })
  );
}

async function runMigrations(db: Client): Promise<void> {
  // 1. تأكد من وجود جدول _migrations
  await db.execute({
    sql: `CREATE TABLE IF NOT EXISTS _migrations (
      id TEXT PRIMARY KEY,
      appliedAt INTEGER NOT NULL
    )`,
    args: [],
  });

  // 2. اقرأ الـ migrations المُطبّقة
  const appliedResult = await db.execute({
    sql: "SELECT id FROM _migrations",
    args: [],
  });
  const appliedSet = new Set(
    appliedResult.rows.map((r) => String(r.id))
  );

  // 3. طبّق الجديدة فقط
  const newlyApplied: string[] = [];
  for (const mig of MIGRATIONS) {
    if (appliedSet.has(mig.id)) continue;
    try {
      await db.execute({ sql: mig.sql, args: [] });
      await db.execute({
        sql: "INSERT OR IGNORE INTO _migrations (id, appliedAt) VALUES (?, ?)",
        args: [mig.id, Date.now()],
      });
      newlyApplied.push(mig.id);
    } catch (err) {
      console.error(`[db] Migration ${mig.id} failed:`, err);
      throw err;
    }
  }

  if (newlyApplied.length > 0) {
    console.log(`[db] Applied ${newlyApplied.length} migrations:`, newlyApplied);
  }
}

export async function ensureDbReady(): Promise<Client> {
  if (globalForDb.__cb_db_ready) return globalForDb.__cb_db_ready;

  const db = getDb();
  const readyPromise = (async () => {
    await runMigrations(db);
    return db;
  })();

  globalForDb.__cb_db_ready = readyPromise;

  try {
    await readyPromise;
  } catch (err) {
    // لو فشلت، اسمح بمحاولة تانية بعدين
    globalForDb.__cb_db_ready = undefined;
    throw err;
  }

  return readyPromise;
}

  const result = await exec(query, args);

export async function getSettings(userId: string): Promise<UserSettings> {
  const row = await firstRow<RawSettings>(
    "SELECT * FROM user_settings WHERE userId = ?",
    [userId]
  );

  if (row) return mapSettings(row);

  const id = generateId();
  const ts = now();

  await exec(
    `INSERT INTO user_settings (id, userId, createdAt, updatedAt)
     VALUES (?, ?, ?, ?)`,
    [id, userId, ts, ts]
  );

  return {
    id,
    userId,
    theme: "dark",
    transcriptFontSize: 16,
    colorCodingEnabled: true,
    soundDetectionEnabled: false,
    soundSensitivityQuiet: 15,
    soundSensitivityNormal: 45,
    soundSensitivityLoud: 70,
    soundSensitivitySpike: 30,
    createdAt: ts,
    updatedAt: ts,
  };
}

export async function updateSettings(
  userId: string,
  patch: Partial<Omit<UserSettings, "id" | "userId" | "createdAt">>
): Promise<UserSettings> {
  const current = await getSettings(userId);

  const updated = {
    theme: patch.theme ?? current.theme,
    transcriptFontSize: patch.transcriptFontSize ?? current.transcriptFontSize,
    colorCodingEnabled: patch.colorCodingEnabled ?? current.colorCodingEnabled,
    soundDetectionEnabled:
      patch.soundDetectionEnabled ?? current.soundDetectionEnabled,
    soundSensitivityQuiet:
      patch.soundSensitivityQuiet ?? current.soundSensitivityQuiet,
    soundSensitivityNormal:
      patch.soundSensitivityNormal ?? current.soundSensitivityNormal,
    soundSensitivityLoud:
      patch.soundSensitivityLoud ?? current.soundSensitivityLoud,
    soundSensitivitySpike:
      patch.soundSensitivitySpike ?? current.soundSensitivitySpike,
  };

  await exec(
    `UPDATE user_settings
     SET theme = ?,
         transcriptFontSize = ?,
         colorCodingEnabled = ?,
         soundDetectionEnabled = ?,
         soundSensitivityQuiet = ?,
         soundSensitivityNormal = ?,
         soundSensitivityLoud = ?,
         soundSensitivitySpike = ?,
         updatedAt = ?
     WHERE userId = ?`,
    [
      updated.theme,
      updated.transcriptFontSize,
      updated.colorCodingEnabled ? 1 : 0,
      updated.soundDetectionEnabled ? 1 : 0,
      updated.soundSensitivityQuiet,
      updated.soundSensitivityNormal,
      updated.soundSensitivityLoud,
      updated.soundSensitivitySpike,
      now(),
      userId,
    ]
  );

  return getSettings(userId);
}

function hashPassword(password: string): { hash: string; salt: string } {
  const salt = randomBytes(16).toString("hex");
  const hash = scryptSync(password, salt, SCRYPT_KEYLEN).toString("hex");
  return { hash, salt };
}

function verifyPasswordHash(
  password: string,
  hash: string,
  salt: string
): boolean {
  try {
    const derived = scryptSync(password, salt, SCRYPT_KEYLEN);
    const stored = Buffer.from(hash, "hex");
    if (derived.length !== stored.length) return false;
    return timingSafeEqual(derived, stored);
  } catch {
    return false;
  }
}

export async function getAdminPassword(): Promise<{
  hash: string;
  salt: string;
} | null> {
  const row = await firstRow<{ passwordHash: string; passwordSalt: string }>(
    "SELECT passwordHash, passwordSalt FROM admin_settings LIMIT 1"
  );

  if (!row) return null;
  return { hash: row.passwordHash, salt: row.passwordSalt };
}

export async function setAdminPassword(password: string): Promise<void> {
  const { hash, salt } = hashPassword(password);
  const ts = now();

  const existing = await firstRow<{ id: string }>(
    "SELECT id FROM admin_settings LIMIT 1"
  );

  if (existing) {
    await exec(
      `UPDATE admin_settings
       SET passwordHash = ?, passwordSalt = ?, updatedAt = ?
       WHERE id = ?`,
      [hash, salt, ts, existing.id]
    );
  } else {
    await exec(
      `INSERT INTO admin_settings (id, passwordHash, passwordSalt, updatedAt)
       VALUES (?, ?, ?, ?)`,
      [generateId(), hash, salt, ts]
    );
  }
}

export async function verifyAdminPassword(password: string): Promise<boolean> {
  const stored = await getAdminPassword();

  if (!stored) {
    const envPassword = process.env.ADMIN_PASSWORD?.trim();
    return !!envPassword && password === envPassword;
  }

  return verifyPasswordHash(password, stored.hash, stored.salt);
}

export async function hasCustomAdminPassword(): Promise<boolean> {
  const row = await firstRow<{ id: string }>(
    "SELECT id FROM admin_settings LIMIT 1"
  );
  return !!row;
}
